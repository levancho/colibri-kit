import { Sparkline } from "./charts-react";
export function Kpi({ label, value, trend, delta, up = true }) {
  return (
    <article className="card kpi">
      <small>{label}</small>
      <b>{value}</b>
      <Sparkline values={trend} label={`${label} trend`} />
      <span className="delta"><span className={up ? "up" : "down"}>{up ? "▲" : "▼"} {delta}</span> vs last month</span>
    </article>
  );
}
<Kpi label="Revenue" value="$48,290" trend={[12,18,14,22,19,27,31]} delta="12.4%" />
<#macro kpi label value trend delta up=true>
  <article class="card kpi">
    <small>${label}</small>
    <b>${value}</b>
    <span data-chart="sparkline" data-values="[${trend?join(",")}]" aria-label="${label} trend"></span>
    <span class="delta"><span class="${up?then('up','down')}">${up?then('▲','▼')} ${delta}</span> vs last month</span>
  </article>
</#macro>

<script type="module">import { mount } from "./charts.js"; mount();</script>
import { useEffect, useRef } from "react";
import { area } from "./charts.js";
export function AreaChart({ series, title, sub }) {
  const ref = useRef(null);
  useEffect(() => { ref.current.innerHTML = area(series, { label: title }); }, [series, title]);
  return (
    <section className="card panel chart-box">
      <header><h3>{title}</h3>{sub && <p>{sub}</p>}</header>
      <div ref={ref} />
    </section>
  );
}
<#macro chart kind title values sub="">
  <section class="card panel chart-box">
    <header><h3>${title}</h3><#if sub?has_content><p>${sub}</p></#if></header>
    <div data-chart="${kind}" data-values='${values}' aria-label="${title}"></div>
  </section>
</#macro>

export function CustomerTable({ rows }) {
  return (
    <table className="data">
      <thead><tr><th>Customer</th><th>Plan</th><th>MRR</th><th>Status</th></tr></thead>
      <tbody>{rows.map((c) => (
        <tr key={c.id}><td><span className="who"><span className="av">{c.initials}</span>{c.name}</span></td>
          <td>{c.plan}</td><td>{c.mrr}</td>
          <td><span className="badge" data-tone={c.active ? "ok" : "warn"}>{c.status}</span></td></tr>
      ))}</tbody>
    </table>
  );
}
